using System;
using GTA;
using GTA.Math;
using GTA.Native;

namespace DeadWastelandXbox
{
    public class MenuRS : Script
    {
        private int ammoCount = 150;
        private int scrapCount = 45;
        private int foodCount = 20;

        public MenuRS()
        {
            Tick += OnTick;
        }

        private void OnTick(object sender, EventArgs e)
        {
            Ped player = Game.Player.Character;

            // Teclado USB: R + S  OR  Controle: RB + D-Pad Baixo
            bool isRS = Game.IsKeyPressed(System.Windows.Forms.Keys.R) && Game.IsKeyPressed(System.Windows.Forms.Keys.S);
            bool isPadMenu = Game.IsControlPressed(0, Control.Sprint) && Game.IsControlPressed(0, Control.PhoneDown);

            if (isRS || isPadMenu)
            {
                UI.Notify("~b~=== DEAD WASTELAND: RÁDIO & STATUS ===");
                UI.Notify($"~w~Líder: ~g~Vida {player.Health} HP ~w~| ~b~Stamina OK");
                UI.Notify($"~w~Recursos: ~g~Comida: {foodCount} ~w~| ~y~Peças: {scrapCount} ~w~| ~r~Munição: {ammoCount}");
                UI.Notify("~w~Pressione ~g~1 ~w~para [Chamar Reforço] via Rádio.");
                
                Script.Wait(1000);
            }

            // Comando Rápido: Chamar Reforço (Teclado: 1)
            if (Game.IsKeyPressed(System.Windows.Forms.Keys.D1))
            {
                CallReinforcements(player);
                Script.Wait(2000);
            }
        }

        private void CallReinforcements(Ped player)
        {
            // Spawna um Patrulheiro aliado próximo ao jogador
            Ped ally = World.CreatePed(PedHash.Hitchhiker, player.Position + player.ForwardVector * 3.0f);
            if (ally != null)
            {
                ally.AddBlip();
                ally.CurrentBlip.Color = BlipColor.Blue;
                ally.CurrentBlip.Name = "Patrulheiro de Reforço 🔵";
                
                // Manda o aliado seguir e proteger o jogador
                Function.Call(Hash.TASK_COMBAT_PED, ally, Game.Player.Character, 0, 16);
                UI.Notify("~b~[RÁDIO] Patrulheiros a caminho da sua localização! 🔵");
            }
        }
    }
}
